Задача 2: «Анализатор последовательности»
javascript
let sum = 0;

for (let i = 1; i <= 200; i++) {
    if (i % 3 === 0 && i % 5 !== 0) {
        sum = sum + i;
    }
}

console.log('Сумма чисел, которые делятся на 3, но не делятся на 5:', sum);
